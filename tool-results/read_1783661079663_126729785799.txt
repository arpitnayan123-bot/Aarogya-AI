   344→      const prevItem = cat.find((b: BiomarkerItem) => b.name_en === currentName);

   349→        const higherIsGood = isHigherBetter(currentName);
   350→        const isImproving = (diff > 0 && higherIsGood) || (diff < 0 && !higherIsGood);
   351→        
   352→        if (isImproving) return { icon: <TrendingDown className={`w-4 h-4 text-emerald-500`} />, text: 'Improving', color: 'text-emerald-600' }; // Down is good for bad things like sugar, Up is good for Hemoglobin. Wait.
   353→        // If Higher is Better (Hemoglobin): Current > Prev -> Good (Green Up). Current < Prev -> Bad (Red Down).
   354→        // If Lower is Better (Glucose): Current < Prev -> Good (Green Down). Current > Prev -> Bad (Red Up).
   355→        
   356→        if (higherIsGood) {
   357→            return diff > 0 
   358→                ? { icon: <TrendingUp className="w-4 h-4 text-emerald-500" />, text: `Was ${prevItem.value} - Improving!`, color: 'text-emerald-600' }
